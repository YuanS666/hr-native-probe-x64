cmake_minimum_required(VERSION 3.20)
project(screen_native_probe LANGUAGES CXX)
enable_testing()
add_executable(bitstream_test bitstream_test.cpp)
target_compile_features(bitstream_test PRIVATE cxx_std_17)
add_test(NAME bitstream COMMAND bitstream_test)
if(WIN32)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Use Visual Studio C++ x64 and Windows SDK")
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../index.html" PROBE_HTML)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/playback.js" PROBE_JS)
    configure_file(page.hpp.in page.hpp @ONLY)
    add_executable(screen_native_probe probe.cpp)
    target_compile_features(screen_native_probe PRIVATE cxx_std_17)
    target_include_directories(screen_native_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(screen_native_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_compile_options(screen_native_probe PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(bitstream_test PRIVATE /W4 /utf-8 /EHsc)
    target_link_libraries(screen_native_probe PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 user32 psapi pdh)
    set_property(TARGET screen_native_probe bitstream_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    message(STATUS "Non-Windows: portable bitstream tests only; native capture is NOT built")
endif()
