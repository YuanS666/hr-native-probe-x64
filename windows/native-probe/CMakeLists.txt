cmake_minimum_required(VERSION 3.20)
project(screen_native_probe LANGUAGES CXX)
enable_testing()
# 网页按UTF-8原始字节生成数组，避免MSVC单个字符串字面量长度限制，并保持目标机无需外部网页文件。
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../index.html" PROBE_HTML HEX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/playback.js" PROBE_JS HEX)
foreach(NAME IN ITEMS PROBE_HTML PROBE_JS)
    string(REGEX REPLACE "(................................)" "\\1\n" ${NAME} "${${NAME}}")
    string(REGEX REPLACE "([0-9a-fA-F][0-9a-fA-F])" "0x\\1," ${NAME}_BYTES "${${NAME}}")
endforeach()
configure_file(page.hpp.in page.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../index.html" "${CMAKE_CURRENT_SOURCE_DIR}/playback.js")
add_executable(page_test page_test.cpp)
target_compile_features(page_test PRIVATE cxx_std_17)
target_include_directories(page_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME embedded_page COMMAND page_test
    "${CMAKE_CURRENT_SOURCE_DIR}/../../index.html" "${CMAKE_CURRENT_SOURCE_DIR}/playback.js")
add_executable(bitstream_test bitstream_test.cpp)
target_compile_features(bitstream_test PRIVATE cxx_std_17)
add_test(NAME bitstream COMMAND bitstream_test)
if(WIN32)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Use Visual Studio C++ x64 and Windows SDK")
    endif()
    add_executable(screen_native_probe probe.cpp)
    target_compile_features(screen_native_probe PRIVATE cxx_std_17)
    target_include_directories(screen_native_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(screen_native_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_compile_options(screen_native_probe PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(bitstream_test PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(page_test PRIVATE /W4 /utf-8 /EHsc)
    target_link_libraries(screen_native_probe PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 user32 psapi pdh)
    set_property(TARGET screen_native_probe bitstream_test page_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    message(STATUS "Non-Windows: portable bitstream and embedded-page tests only; native capture is NOT built")
endif()
